import numpy as np

from sklearn.ensemble import GradientBoostingClassifier, StackingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline

from services.deployment_service import (
    DeploymentError,
    _validate_stacking_artifact,
    stacking_deployment_eligibility,

    STACKING_MODEL_ID,
    STACKING_MODEL_NAME,
    build_individual_estimators,
)
from services.preprocessing_service import build_feature_preprocessor

V3, V4 = "stacking-five-v3", "stacking-five-v4"


def _artifact(version, old_gradient_boosting):
    rng = np.random.default_rng(7)
    values = rng.normal(size=(60, 2))
    features = pd.DataFrame(values, columns=["a", "b"])
    labels = pd.Series((values[:, 0] + values[:, 1] > 0).astype(int))
    members = build_individual_estimators(n_neighbors=3)
    if old_gradient_boosting:
        members["gradient_boosting"] = GradientBoostingClassifier(random_state=42)
    stack = StackingClassifier(
        estimators=list(members.items()),
        final_estimator=LogisticRegression(max_iter=1000),
        stack_method="predict_proba",
        cv=2,
    )
    pipeline = Pipeline(
        [("preprocessor", build_feature_preprocessor(["a", "b"], [])), ("classifier", stack)]
    )
    pipeline.fit(features, labels)
    return {"model_id": STACKING_MODEL_ID, "model_name": STACKING_MODEL_NAME,
            "workflow_version": version, "pipeline": pipeline}


@pytest.mark.parametrize("version, old_member", [(V3, True), (V4, False)])
def test_each_version_is_accepted_with_its_own_gradient_boosting_member(version, old_member):

def test_a_member_from_the_other_version_is_refused(version, old_member):

```python
"""The decision line of a Stacking Ensemble, carried inside the model (DEF-24, training run 13).

AlgoGuard calls a flow an attack when the Attack probability is the higher of the two, that is at
0.50. To use another line X without changing the app, the Logistic Regression that combines the
five members has its intercept lowered by logit(X). The new score, sigmoid(z - logit(X)), is
exactly 0.50 where the old probability was X. The order of the scores, and so the ROC-AUC, does
not change.
"""

import math

from sklearn.ensemble import StackingClassifier
from sklearn.linear_model import LogisticRegression

DEFAULT_DECISION_LINE = 0.5


def validate_decision_line(line: float | str) -> float:
    """Return ``line`` as a float strictly between 0 and 1, or raise ``ValueError``."""
    try:
        value = float(line)
    except (TypeError, ValueError):
        value = math.nan
    if not math.isfinite(value) or not 0.0 < value < 1.0:
        raise ValueError("The decision line must be between 0 and 1, for example 0.30.")
    return value


def apply_decision_line(stacking, line):
    """Set ``line`` inside a fitted Stacking Ensemble and return it as a float."""
    value = validate_decision_line(line)
    if not isinstance(stacking, StackingClassifier):
        raise TypeError("The decision line applies to a Stacking Ensemble only.")
    combiner = getattr(stacking, "final_estimator_", None)
    if not isinstance(combiner, LogisticRegression):
        raise TypeError("The Stacking Ensemble must be fitted with a Logistic Regression combiner.")
    if value != DEFAULT_DECISION_LINE:
        combiner.intercept_ = combiner.intercept_ - math.log(value / (1.0 - value))
    return value
```

def key_groups(frame):
    """One group id per 13-feature key, the same in every process (unlike Python's hash())."""
    norm = pd.DataFrame(index=frame.index)
    for column in KEY:
        values = frame[column]
        if pd.api.types.is_numeric_dtype(values):
            norm[column] = values.astype(float).round(9)
        else:
            norm[column] = values.astype(str).str.strip()
    return pd.util.hash_pandas_object(norm, index=False).to_numpy()


def recording_flows(path):
    """The 16 features of every flow in a recording, derived by the app's own flow tracker."""
    tracker, flows = FlowTracker(), []
    with PcapReader(str(path)) as reader:
        for packet in reader:
            info = packet_info_from_scapy(packet)
            if info is not None:
                flows.extend(tracker.add_packet(info))
    flows.extend(tracker.flush())
    return pd.DataFrame([flow["features"] for flow in flows], columns=list(FLOW_FEATURE_COLUMNS))


def rates(y, scores, line):
    """Accuracy, F1, recall and false-positive rate when scores at or above ``line`` are attacks."""
    y = np.asarray(y).astype(int)
    pred = np.asarray(scores) >= line
    tp = int((pred & (y == 1)).sum())
    fp = int((pred & (y == 0)).sum())
    fn = int((~pred & (y == 1)).sum())
    tn = int((~pred & (y == 0)).sum())
    recall = tp / max(tp + fn, 1)
    precision = tp / max(tp + fp, 1)
    return {"accuracy": (tp + tn) / max(len(y), 1),
            "f1": 2 * precision * recall / max(precision + recall, 1e-12),
            "recall": recall, "fpr": fp / max(fp + tn, 1)}


def attack_scores(pipeline, frame, columns):
    """The Attack column of ``predict_proba`` (found by class value, not position)."""
    probabilities = np.asarray(pipeline.predict_proba(frame[list(columns)]))
    return probabilities[:, list(pipeline.classes_).index(1)]

    for fold, (train_index, test_index) in enumerate(splits, start=1):
        fold_training = pd.concat([others, file2.iloc[train_index]], ignore_index=True)
        pipeline = stacking_pipeline(fold_training)
        pipeline.fit(fold_training[COLUMNS], fold_training["label"].astype(int))
        out_of_fold[test_index] = attack_scores(pipeline, file2.iloc[test_index], COLUMNS)
        holdout_scores += attack_scores(pipeline, holdout, COLUMNS) / FOLDS
        for name, frame in recordings.items():
            recording_scores[name] += attack_scores(pipeline, frame, COLUMNS) / FOLDS
        print(f"fold {fold}/{FOLDS} done after {time.time() - started:.0f} s", flush=True)

    y_file2 = file2["label"].astype(int).to_numpy()
    y_holdout = holdout["label"].astype(int).to_numpy()
    fuzzers = file2_categories.to_numpy() == "Fuzzers"
    sweep = []
    for line in LINES:
        raw = rates(y_file2, out_of_fold, line)
        browsing = {name: float((scores >= line).mean())
                    for name, scores in recording_scores.items()}
        row_margin = margin(raw, max(browsing.values()))
        sweep.append({
            "line": line, **{key: round(value, 4) for key, value in raw.items()},
            "fuzzers_recall": round(float((out_of_fold[fuzzers] >= line).mean()), 4),
            "browsing": {name: round(value, 4) for name, value in browsing.items()},
            "browsing_max": round(max(browsing.values()), 4),
            "holdout": {key: round(value, 4)
                        for key, value in rates(y_holdout, holdout_scores, line).items()},
            "margin": round(row_margin, 4), "qualifies": row_margin >= 0,
        })
    qualifying = [row for row in sweep if row["qualifies"]]
    chosen = max(qualifying, key=lambda row: (row["margin"], row["line"])) if qualifying else None
    report = {
        "protocol": "test-results/model_run13/protocol_run13.md",
        "workflow_version": MODEL_WORKFLOW_VERSION, "seed": SEED, "folds": FOLDS,
        "file2_rows": len(file2), "file2_auc_out_of_fold": round(
            float(roc_auc_score(y_file2, out_of_fold)), 4),
        "sweep": sweep, "chosen_line": chosen["line"] if chosen else None,
        "stop": chosen is None, "minutes": round((time.time() - started) / 60, 1),
    }
    (HERE / "line_choice.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    for row in sweep:
        print(f"{row['line']:.2f} acc {row['accuracy']:.4f} f1 {row['f1']:.4f} "
              f"rec {row['recall']:.4f} fpr {row['fpr']:.4f} browse {row['browsing_max']:.4f} "
              f"ho-fpr {row['holdout']['fpr']:.4f} margin {row['margin']:+.4f}")
    print("chosen line:", report["chosen_line"])

    main()

# Training run 13: protocol (written before any run 13 data was built or model trained)

Written <NOW> (Asia/Manila) and committed before `build_run13_data.py` was run. The code is that

## Why

DEF-24: model 60 catches 76.0% of attacks on unseen UNSW-NB15 rows (target 90%). 80% of its misses
are Fuzzers of the official training partition ("file 2").

The 2026-11-17 check found that the following meets every target on a held slice of file 2, and
keeps PC A's browsing recordings within 8%:
- train on both files;
- weight PC A's benign flows 10x;
- use a histogram Gradient Boosting member;
- set a lower decision line.

## Data roles

| Role | Data |
|---|---|
| Training | `research/dataset/processed/run13/training.csv`: run 11's training file plus the PC A flows 9 more times, 167,082 rows. Run 11's file holds 66,393 file-1 rows, 2,460 PC A benign flows and 76,089 file-2 rows. Built by `build_run13_data.py`; SHA-256 in `run13_data_manifest.json` |
| Line choice (selection) | Out-of-fold scores on the 76,089 file-2 rows: `StratifiedGroupKFold(3, shuffle=True, random_state=13)`, groups = the 13-feature key hashed with `pandas.util.hash_pandas_object`, strata = attack category. Fold averages on the hold-out and the four selection recordings |
| Selection recordings | `research/dataset/processed/def19/generator.pcap`, `browser.pcap`, `final_browser.pcap`, `final_unseen.pcap` |
| Hold-out | `research/dataset/processed/flow_window_v2/holdout_eval.csv` (file 1, 15,939 rows); reported, not gated |
| Fresh browsing | Two 120-second recordings on PC A made after the final model exists |
| Final test | `research/dataset/processed/run11/final_test.csv`: 32,610 rows of file 2, SHA-256 in `../model_run11/run11_data_manifest.json`. No model has read it. Scored once |

## Steps

1. **Line choice** (`choose_line.py`).
   - Lines 0.20 to 0.50 in steps of 0.05.
   - A line qualifies if the out-of-fold file-2 accuracy, F1 and recall are at least 0.90, the
     FPR is at most 0.08, and every selection recording is at most 8% Attack.
   - Choose the qualifying line with the largest smallest margin over those five limits; on a
     tie, the higher line.
   - If no line qualifies, stop: no model is trained.
2. **Final model.** `python train.py research\dataset\processed\run13\training.csv
   --decision-line <line>`.
   - One candidate: no tuning and no second try.
   - If the process is interrupted, the same command is run again unchanged, as run 11 became

   - Each selection recording at most 8% Attack.
   - Artifact at most 50 MiB.

   - The artifact carries the chosen line and workflow v4.
   - The hold-out is reported, not gated. Its FPR is expected at about 11-15% (run 10: 2.7%).
4. **Fresh browsing** (`score_fresh.py`).
   - Two recordings: the generator sites, and `--sites unseen`.
   - Each needs at least 50 flows (fewer: recorded again, not counted) and at most 8% Attack by
     run 13. Model 60 is scored for comparison.
5. **Final test, once** (`final_eval_run13.py`).
   - `scripts/evaluate_model.py` with run 13's training file as `--train-csv`; run 13 and model 60

   - Targets: accuracy, F1 and recall at least 0.90; FPR at most 0.08; ROC-AUC at least 0.95.
   - Also reported: recall per attack category, Brier score, ECE (10 bins) and score bands.
6. **Ship rule.** Run 13 ships only if steps 3 and 4 passed and all five targets are met in

   - commits <CODE_FIRST> and <CODE_COMMIT> are reverted;

- **Live evidence is benign traffic on one PC.** The selection recordings also chose the line, so

- **Fold models see more file-1 rows** than the final model: `train.py` keeps 25% of rows back as
  its check split.
